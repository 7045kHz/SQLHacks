```sql
SELECT *
		FROM INFORMATION_SCHEMA.COLUMNS
		WHERE TABLE_NAME = N'Customers'
```

## Example

```sql
SELECT	
	COLUMN_NAME , 
	DATA_TYPE = CASE	
		WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL   AND IS_NULLABLE = 'NO' THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ') NOT NULL ,'
		WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL   AND IS_NULLABLE = 'YES'   THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ')  NULL ,'

		WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL    AND IS_NULLABLE = 'NO'     THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ') NOT NULL ,'
		WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL    AND IS_NULLABLE = 'YES'     THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ')  NULL ,'

		WHEN  CHARACTER_MAXIMUM_LENGTH  IS NULL    AND IS_NULLABLE = 'NO'  THEN    CONVERT(varchar(200),DATA_TYPE) + ' NOT NULL,'
		WHEN  CHARACTER_MAXIMUM_LENGTH  IS NULL   AND IS_NULLABLE = 'YES'   THEN    CONVERT(varchar(200),DATA_TYPE) + ' NULL,'

	ELSE  CONVERT(varchar(200),DATA_TYPE) + ','
			END
			
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = N'EXCLUSION_DATES'
AND TABLE_SCHEMA = N'CTRL'
```

## Stored Procedure Example

```sql
CREATE PROCEDURE [CTRL].[CreateTempTableFromTable]
@SchemaName nvarchar(250),
@TableName nvarchar(250)
AS
BEGIN
DECLARE @SQL nvarchar(4000)
DECLARE @SQL_T_NAME nvarchar(250)
DECLARE @SQL_T_SELECT nvarchar(250)
DECLARE @col nvarchar(250)
DECLARE @typ nvarchar(250)

SET @SQL_T_NAME = N'#' + @SchemaName +'_' + @TableName +' '; 
SET @SQL_T_SELECT= N'SELECT * FROM ' + @SQL_T_NAME + N''; 
SET @SQL = N'CREATE TABLE '+ @SQL_T_NAME +' ('; 

DECLARE db_cursor CURSOR 
	FOR	SELECT	
			COLUMN_NAME as col, 
			typ = CASE	
				WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL   AND IS_NULLABLE = 'NO'   THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ') NOT NULL ,'
				WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL   AND IS_NULLABLE = 'YES'   THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ')  NULL ,'

				WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL    AND IS_NULLABLE = 'NO'     THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ') NOT NULL ,'
				WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL    AND IS_NULLABLE = 'YES'     THEN CONVERT(varchar(200),DATA_TYPE) + '(' + CONVERT(varchar(200),CHARACTER_MAXIMUM_LENGTH)  + ')  NULL ,'

				WHEN  CHARACTER_MAXIMUM_LENGTH  IS NULL    AND IS_NULLABLE = 'NO'  THEN    CONVERT(varchar(200),DATA_TYPE) + ' NOT NULL,'
				WHEN  CHARACTER_MAXIMUM_LENGTH  IS NULL   AND IS_NULLABLE = 'YES'   THEN    CONVERT(varchar(200),DATA_TYPE) + ' NULL,'

				ELSE  CONVERT(varchar(200),DATA_TYPE) + ','
			END
			
		FROM INFORMATION_SCHEMA.COLUMNS
		WHERE TABLE_NAME = @TableName
		AND TABLE_SCHEMA = @SchemaName

		OPEN db_cursor  
FETCH NEXT FROM db_cursor INTO @col, @typ  
WHILE @@FETCH_STATUS = 0  
BEGIN  
      SET @SQL = @SQL +N'    ' +  @col +N'    ' +  @typ
      FETCH NEXT FROM db_cursor INTO  @col, @typ  
END 

CLOSE db_cursor  
DEALLOCATE db_cursor

SET @SQL =  STUFF(@SQL, LEN(@SQL), 1, ');')
EXEC sp_executesql  @SQL;
EXEC sp_executesql  @SQL_T_SELECT; 
END;
```
